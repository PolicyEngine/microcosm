def audit(event, args):
    if event == "subprocess.Popen":
        executable, argv, cwd, _environment = args
        allowed = (["git", "ls-files", "--", "packages/*/tests/test_*.py"], ["git", "ls-files", "--", "packages"])
        if executable == "git" and argv in allowed and pathlib.Path(cwd) == BASE and len(GIT_CALLS) < 2:
            GIT_CALLS.append(argv); return
    if event in ('os.listdir', 'os.scandir') and args and isinstance(args[0], (str, bytes)) and not any(pathlib.Path(os.path.abspath(os.fsdecode(args[0]))).is_relative_to(root) for root in (*CODE_ROOTS, RUN)):
        BLOCKED.append('directory:' + str(args[0])); raise PermissionError('Unrelated scratch enumeration refused')
    if (event.startswith('socket.') and event != 'socket.gethostname') or event in {'subprocess.Popen','os.system','os.posix_spawn','os.exec','os.fork','pty.spawn'}:
        BLOCKED.append(event)
        raise PermissionError('External operation blocked: ' + event)
    if event == 'open' and isinstance(args[0], (str, bytes)):
        p = pathlib.Path(os.path.abspath(os.fsdecode(args[0])))
        if str(p).endswith(('.err','.lane.log')):
            BLOCKED.append(str(p)); raise RuntimeError('Forbidden log')
        if p.is_relative_to(RUN): return
        if p == EXECUTABLE: return
        if p in ALLOW:
            READS.add(str(p)); return
        if p == pathlib.Path('/dev/null'): return
        if any(p.is_relative_to(root) for root in CODE_ROOTS):
            if p.suffix in {'.py','.pyc','.so','.dylib','.css','.js','.html','.md','.toml'}: return
            if p.name in {'METADATA','WHEEL','entry_points.txt','top_level.txt','py.typed'} and any(part.endswith('.dist-info') for part in p.parts): return
        BLOCKED.append(str(p))
        raise PermissionError('Read blocked before open: ' + str(p))
# Metadata calls do not emit Python audit events. Deny non-code paths before stat.
_STAT, _LSTAT = os.stat, os.lstat
def metadata_allowed(value):
    if isinstance(value, int): return True
    p = pathlib.Path(os.path.abspath(os.fsdecode(value)))
    if str(p).endswith(('.err', '.lane.log')): return False
    if p.name == 'RECORD' and p not in ALLOW: return False
    if any(token in str(p).lower() for token in ("b19001", "b25003", "occupied-hu", "occupied_hu")): return False
    if p.is_relative_to(BASE/'packages') and 'tests' in p.relative_to(BASE/'packages').parts: return True
    if p == EXECUTABLE or EXECUTABLE.is_relative_to(p): return True
    if p.is_relative_to(RUN) or p in ALLOW or p == pathlib.Path(sys.base_prefix)/'lib/python3.14': return True
    if any(root.is_relative_to(p) for root in (*CODE_ROOTS, RUN)): return True
    if any(p.is_relative_to(root) for root in CODE_ROOTS):
        return (p.suffix in {'', '.py', '.pyc', '.so', '.dylib', '.css', '.js', '.html', '.md', '.toml', '.dist-info', '.git'}
                or p.name in {'METADATA', 'WHEEL', 'entry_points.txt', 'top_level.txt', 'py.typed'})
    return False
def guarded_stat(path, *args, **kwargs):
    if not metadata_allowed(path):
        BLOCKED.append('stat:' + os.fsdecode(path))
        raise PermissionError('Metadata blocked before stat')
    return _STAT(path, *args, **kwargs)
def guarded_lstat(path, *args, **kwargs):
    if not metadata_allowed(path):
        BLOCKED.append('lstat:' + os.fsdecode(path))
        raise PermissionError('Metadata blocked before lstat')
    return _LSTAT(path, *args, **kwargs)
os.stat, os.lstat = guarded_stat, guarded_lstat
